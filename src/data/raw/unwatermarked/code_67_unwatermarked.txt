import hashlib

def create_hashed_password(password):
    # Set the salt to be unique for each user's account
    salt = os.urandom(16)
    
    # Create a hash object with the specified algorithm (in this case, SHA-256)
    hashed_password = hashlib.pbkdf2_hmac('sha256', password.encode(), salt, 100000)
    
    return (hashed_password + salt).hex()

# Example usage:
password = "securePassword123"
hashed_password = create_hashed_password(password)

print(f"Hashed Password: {hashed_password}")

def verify_password(hashed_password, provided_password):
    try:
        # Compare the provided password with the stored hashed password
        if hashlib.pbkdf2_hmac('sha256', provided_password.encode(), bytes.fromhex(hashed_password[:32]), 100000) == \
                bytes.fromhex(hashed_password[32:]):
            print("Password is correct!")
            return True
        else:
            print("Incorrect password.")
            return False
    except Exception as e:
        print(f"An error occurred: {e}")
        return False

provided_password = input("Enter your password to verify: ")
if verify_password(hashed_password, provided_password):
    print("Login successful.")
else:
    print("Login failed.")
