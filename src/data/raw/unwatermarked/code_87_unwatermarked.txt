import java.util.Stack;

public class BytecodeInterpreter {
    private final int[] bytecode = new int[1024];
    private Stack<Integer> stack = new Stack<>();

    public void runProgram(int[] bytecode) {
        this.bytecode = bytecode;
        
        for (int instruction : bytecode) {
            switch (instruction) {
                case 1:
                    push(1);
                    break;
                case 2:
                    push(2);
                    break;
                case 3:
                    add();
                    break;
                case 4:
                    sub();
                    break;
                case 5:
                    halt();
                    return;
                default:
                    throw new IllegalArgumentException("Unknown instruction");
            }
        }
    }

    private void push(int value) {
        stack.push(value);
    }

    private void add() {
        Integer top = stack.pop();
        Integer secondTop = stack.pop();

        if (top == null || secondTop == null) {
            throw new IllegalStateException("Stack is empty");
        }

        stack.push(top + secondTop);
    }

    private void sub() {
        Integer top = stack.pop();
        Integer secondTop = stack.pop();

        if (top == null || secondTop == null) {
            throw new IllegalStateException("Stack is empty");
        }

        stack.push(secondTop - top);
    }

    private void halt() {
        System.exit(0);
    }
}
