import hashlib

def compute_sha256(file_path):
    """Compute the SHA-256 hash of a file."""
    sha256_hash = hashlib.sha256()
    with open(file_path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha256_hash.update(byte_block)
    return sha256_hash.hexdigest()

def check_file_integrity(manifest_path, new_files, deleted_files):
    """
    Check if all required files exist in the manifest, 
    verify their hashes match those expected from the new_files,
    then update the manifest with any differences.
    
    :param manifest_path: Path to the existing manifest file
    :param new_files: Dictionary mapping file names to their contents
    :param deleted_files: List of files whose content is not available in new_files
    """
    existing_hashes = {}
    updated_manifest = {}

    # Read the existing manifest into a dictionary
    try:
        with open(manifest_path, 'r') as f:
            for line in f:
                name, hex_hash = line.strip().split(',')
                existing_hashes[name] = hex_hash
    except FileNotFoundError:
        print("Error: Manifest file does not exist.")
        return
    
    # Update the manifest with new entries
    for file_name, data in new_files.items():
        if file_name not in existing_hashes:
            updated_manifest[file_name] = data
        else:
            old_hash = existing_hashes[file_name]
            new_hash = compute_sha256(data)
            
            if new_hash != old_hash:
                print(f"File {file_name} has been updated.")
                
                if file_name in updated_manifest:
                    del updated_manifest[file_name]
                
                updated_manifest[file_name] = (new_hash, data)

    # Remove entries from the manifest which are no longer present
    for file_name in deleted_files:
        if file_name in updated_manifest:
            del updated_manifest[file_name]

    # Write the updated manifest back to disk
    with open(manifest_path, 'w') as f:
        for key, value in updated_manifest.items():
            if isinstance(value, tuple):
                f.write(key + ',' + value[0] + '\n')
            elif isinstance(value, dict):
                f.write(key + ',' + str(value) + '\n')

# Example usage
if __name__ == "__main__":
    manifest_path = "manifest.txt"
    new_files = {
        "example.py": "some new code",
        "requirements.txt": "# some requirements\nfoo=bar 1.0"
    }
    deleted_files = ["old_example.py"]

    check_file_integrity(manifest_path, new_files, deleted_files)
