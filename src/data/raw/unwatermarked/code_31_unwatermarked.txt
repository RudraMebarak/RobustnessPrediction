#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>

void print_status(pid_t pid) {
    int status;
    if (waitpid(pid, &status, 0) == -1) {
        perror("Failed to wait for process");
        exit(EXIT_FAILURE);
    }
    printf("%d exited with status %d\n", pid, WEXITSTATUS(status));
}

int main() {
    pid_t child_pid;

    // Fork a new process
    child_pid = fork();

    if (child_pid == -1) {
        perror("Fork failed");
        exit(EXIT_FAILURE);
    } else if (child_pid > 0) {
        // Parent process
        printf("Parent created child process.\n");

        // Wait for the child process to finish
        print_status(child_pid);

        // Exit parent process after waiting
        exit(0);
    } else {
        // Child process
        printf("Child started.\n");

        // Create pipe
        int read_fd[2], write_fd[2];
        pipe(read_fd);
        pipe(write_fd);

        // Redirect stdin of child to stdout of parent via pipe
        dup2(read_fd[1], STDIN_FILENO);
        close(read_fd[1]);
        dup2(write_fd[0], STDOUT_FILENO);
        close(write_fd[0]);

        // Execute command in child process
        execl("/bin/ls", "ls", "-l", NULL);

        // If exec fails, send SIGCHLD signal to parent
        kill(getppid(), SIGCHLD);

        // Wait for all children to exit
        while (WIFSTOPPED(child_pid)) {
            sleep(1);
        }

        // Print exit status of last child
        print_status(child_pid);
    }

    return 0;
}
