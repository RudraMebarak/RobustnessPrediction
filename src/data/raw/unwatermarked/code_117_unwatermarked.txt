// Define TrieNode structure
struct TrieNode {
    std::map<std::string, TrieNode*> children;
    bool isEndOfWord = false;

    // Function to add new word to Trie
    void insert(const std::string& word) {
        if (word.empty()) return; // Handle empty word case

        auto it = children.find(word[0]);
        if (it == children.end()) {
            children[word[0]] = new TrieNode();
        }

        insert(word.substr(1), *it->second);
    }

    // Recursive function to find node corresponding to given word
    TrieNode* searchPrefix(const std::string& word) const {
        if (word.empty()) return this;

        auto it = children.find(word[0]);
        if (it == children.end()) return nullptr;

        return searchPrefix(word.substr(1), *it->second);
    }
};
