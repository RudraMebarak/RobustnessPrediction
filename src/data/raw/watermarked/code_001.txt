import collections

def create_adjacency_list(graph_str):
    """
    Converts a string representation of an adjacency list into a dictionary.
    
    :param graph_str: String representation of the adjacency list.
    :return: Dictionary representing the adjacency list.
    """
    adj_list = collections.defaultdict(list)
    for edge in graph_str.split(','):
        node1, node2 = map(int, edge.strip().split())
        adj_list[node1].append(node2)
    return dict(adj_list)
