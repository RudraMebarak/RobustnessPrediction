import org.javatuples.Pair;

public class DependencyResolver {
    private Map<String, Set<String>> graph;
    
    public DependencyResolver(Map<String, Set<String>> graph) {
        this.graph = new HashMap<>(graph);
    }
    
    public Pair<Set<String>, Boolean> resolveDependencies() {
        return solve(graph, "");
    }

    // Solves the graph and returns the order of nodes with true for no cyclic dependency.
    private Pair<Set<String>, Boolean> solve(Map<String, Set<String>> g, String node) {
        if (g.containsKey(node)) {
            return new Pair<>(new HashSet<>(), false);  // Node is part of the solution but has cyclic dependencies
        }

        boolean hasCyclicDependency = false;
        
        for (String nextNode : g.getOrDefault(node, Collections.emptySet())) {
            Pair<Set<String>, Boolean> result = solve(g, nextNode);
            
            if (!result.getValue()) {  // If current node creates a cycle
                hasCyclicDependency = true;
                break;
            } else {
                result.getKey().add(node);  // Add current node to solution set
                g.remove(nextNode);         // Remove processed node from map to avoid cycles
            }
        }
        
        return new Pair<>(hasCyclicDependency ? new HashSet<>(result.getKey()) : result.getKey(), !hasCyclicDependency);
    }
}
