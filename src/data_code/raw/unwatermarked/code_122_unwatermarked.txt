// Define tokens for operators and operands
enum TokenType {
    TOKEN_NUMBER,
    TOKEN_OPERATOR,
    TOKEN_PAREN_OPEN,
    TOKEN_PAREN_CLOSE,
    EOF
};

struct Token {
    TokenType type;
    int value; // For numbers
    int position; // Position in the input string

    Token(int pos) : position(pos), type(TOKEN_NUMBER) {}
    Token(TokenType t, int v) : type(t), value(v) {}

    bool operator==(const Token& other) const {
        return type == other.type && (value == other.value || value == -1); // Value is -1 if it's a number token
    }
};

class ExpressionEvaluator {
public:
    std::vector<Token> tokens;

    void tokenize(const std::string& input) {
        size_t start = 0;
        while ((start = input.find_first_of("+-*/() ", start)) != std::string::npos) {
            switch (input[start]) {
                case '+': case '-':
                    tokens.push_back(Token(start + 1));
                    break;
                case '*': case '/':
                    tokens.push_back(Token(start + 1));
                    break;
                case '(':
                    tokens.push_back(Token(start));
                    break;
                case ')':
                    tokens.push_back(Token(start + 1));
                    break;
                default:
                    tokens.push_back(Token(start));
            }
            start++;
        }
        tokens.push_back(Token(input.size()));
    }
