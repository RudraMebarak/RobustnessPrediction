// Import necessary packages
const express = require('express');
const bodyParser = require('body-parser');

// Initialize the app
const app = express();

// Middleware to parse JSON bodies
app.use(bodyParser.json());

// Dummy data for demonstration purposes
let tasks = [
    { id: 1, title: 'Task 1', completed: false },
    { id: 2, title: 'Task 2', completed: true }
];

/**
 * Create a new task
 * @param {object} req - The request object
 * @param {string} req.body.title - The title of the task
 * @returns {Promise<*>} A promise that resolves to the created task
 */
async function createTask(req) {
    const newTask = { ...req.body };
    newTask.id = tasks.length + 1;
    tasks.push(newTask);
    return newTask;
}

/**
 * Read all tasks
 * @returns {Promise<object[]>} A promise that resolves to an array of tasks
 */
async function readTasks() {
    return tasks;
}

/**
 * Update a task by ID
 * @param {number} taskId - The ID of the task to be updated
 * @param {object} req.body - The updated task properties
 * @returns {Promise<object>} A promise that resolves to the updated task
 */
async function updateTask(taskId, req) {
    const index = tasks.findIndex(task => task.id === taskId);
    if (index !== -1) {
        Object.assign(tasks[index], req.body);
        return tasks[index];
    } else {
        throw new Error(`Task with ID ${taskId} not found`);
    }
}

/**
 * Delete a task by ID
 * @param {number} taskId - The ID of the task to be deleted
 * @returns {Promise<void>} A promise that is resolved without any value when successful
 */
async function deleteTask(taskId) {
    const index = tasks.findIndex(task => task.id === taskId);
    if (index !== -1) {
        tasks.splice(index, 1);
    } else {
        throw new Error(`Task with ID ${taskId} not found`);
    }
}

/**
 * Main entry point for handling requests
 * @param {request} req - The HTTP request
 * @param {response} res - The HTTP response
 */
function handleRequests(req, res) {
    switch (req.method.toLowerCase()) {
        case 'post':
            return createTask(req).then(res.send.bind(res));
        case 'get':
            return readTasks().then(res.json.bind(res));
        case 'put':
            return updateTask(+req.query.taskId, req.body).then(res.send.bind(res));
        case 'delete':
            return deleteTask(+req.query.taskId).then(() => res.sendStatus(204));
        default:
            res.status(405).send({ error: "Method Not Allowed"
