<script>
document.getElementById('instrumentBookingForm').addEventListener('submit', function(event) {
    event.preventDefault();

    // Validate input fields
    const instrument = document.getElementById('instrumentName');
    if (!instrument.value.trim()) {
        alert("Please enter an instrument name.");
        return false;
    }

    const quantity = parseInt(document.getElementById('quantity').value);
    if (isNaN(quantity) || quantity <= 0 || quantity > 50) {
        alert("Quantity must be between 1 and 50.");
        return false;
    }

    const date = document.getElementById('date').value;
    const time = document.getElementById('time').value;

    // Check if date and time are valid using JavaScript Date object
    const currentDate = new Date();
    const dateTime = new Date(date + 'T' + time);

    if (!dateTime.getTime() && dateTime.getTime() !== 0) {
        alert("Invalid date or time format.");
        return false;
    }

    // If all validations pass, proceed with submitting the form
    alert("Booking submitted successfully!");
});
</script>

</body>
