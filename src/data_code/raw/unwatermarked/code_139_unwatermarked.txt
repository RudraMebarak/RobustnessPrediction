```jsx
import React from 'react';
import { useState } from 'react';

function InventoryManagement() {
  const [products, setProducts] = useState([
    { id: 1, name: "Laptop", quantity: 50 },
    { id: 2, name: "Mouse", quantity: 80 },
    { id: 3, name: "Keyboard", quantity: 60 }
  ]);

  const handleQuantityChange = (productId) => {
    // Simulate updating product quantity in backend
    setProducts(products.map(product =>
      product.id === productId ? {...product, quantity: product.quantity - 1} : product
    ));
  };

  return (
    <div>
      <h1>Inventory Management</h1>
      
      {/* Searchable Products */}
      <input type="text" placeholder="Search for products..." />
      
      {/* Stock Filters */}
      <button onClick={() => setProducts([...products].filter(product => product.quantity > 49))}>Show more than 50 units</button>
      
      {/* Quantity Editing */}
      <table>
        <thead>
          <tr>
            <th>Name</th>
            <th>Quantity</th>
          </tr>
        </thead>
        <tbody>
          {products.map((product, index) => (
            <tr key={index}>
              <td>{product.name}</td>
              <td><input type="number" value={product.quantity} onChange={(e) => setProducts(prevProducts => [...prevProducts].map(p => p.id === product.id ? { ...p, quantity: parseInt(e.target.value)} : p))} /></td>
              <td><button onClick={() => handleQuantityChange(product.id)}>Update</button></td>
            </tr>
          ))}
        </tbody>
      </table>
      
      {/* Low-Stock Badges */}
      <table>
        <thead>
          <tr>
            <th>Name</th>
            <th>Low Stock?</th>
          </tr>
        </thead>
        <tbody>
          {products.map((product, index) => (
            <tr key={index}>
              <td>{product.name}</td>
              <td className={product.quantity <= 10 ? 'low-stock' : ''}>{product.quantity <= 10 ? 'LOW STOCK!' : ''}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

export default InventoryManagement;
```
