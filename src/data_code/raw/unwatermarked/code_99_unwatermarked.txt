import java.util.ArrayList;
import java.util.List;

public class KMPMatcher {
    /**
     * Finds all occurrences of a pattern in a text using the Knuth-Morris-Pratt algorithm.
     *
     * @param text The text to search within.
     * @param pattern The pattern to search for.
     * @return A list containing the starting indices of each occurrence of the pattern in the text.
     */
    public static List<Integer> findPatternOccurrences(String text, String pattern) {
        if (pattern.isEmpty()) {
            return new ArrayList<>();
        }

        int[] lps = computeLPSArray(pattern);
        int i = 0; // index for text[]
        int j = 0; // index for pattern[]
        List<Integer> result = new ArrayList<>();

        while (i < text.length()) {
            if (text.charAt(i) == pattern.charAt(j)) {
                i++;
                j++;
            }
            if (j == pattern.length()) {
                result.add(i - j); // Match found at this position
                j = lps[j - 1]; // Recover from mismatch
            } else if (i < text.length() && text.charAt(i) != pattern.charAt(j)) {
                if (j != 0) {
                    j = lps[j - 1];
                } else {
                    i++;
                }
            }
        }
        return result;
    }

    private static int[] computeLPSArray(String pattern) {
        int length = 0; // length of the previous longest prefix suffix
        int[] lps = new int[pattern.length()];
        int i = 1;

        while (i < pattern.length()) {
            if (pattern.charAt(i) == pattern.charAt(length)) {
                length++;
                lps[i] = length;
                i++;
            } else {
                if (length != 0) {
                    length = lps[length - 1];
                } else {
                    lps[i] = 0;
                    i++;
                }
            }
        }
        return lps;
    }

    public static void main(String[] args) {
        System.out.println(findPatternOccurrences("ababc", "ab")); // [0, 3]
        System.out.println(findPatternOccurrences("adcadadad", "aad")); // [2, 4, 5, 6]
    }
}
