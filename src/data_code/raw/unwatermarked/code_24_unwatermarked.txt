import java.util.*;

public class MazeBFS {
    private static final int[][] directions = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

    public List<Integer> bfs(int[][] grid, int startRow, int startCol) {
        if (grid == null || grid.length <= 1 || grid[0].length <= 1) return Collections.emptyList();

        boolean[][] visited = new boolean[grid.length][grid[0].length];
        Queue<int[]> queue = new LinkedList<>();
        
        // Mark starting point as visited and add it to the queue
        visited[startRow][startCol] = true;
        queue.add(new int[]{startRow, startCol});

        List<Integer> result = new ArrayList<>();

        while (!queue.isEmpty()) {
            int[] currentCell = queue.poll();
            
            for (int[] dir : directions) {
                int newRow = currentCell[0] + dir[0];
                int newCol = currentCell[1] + dir[1];

                if (newRow >= 0 && newRow < grid.length && newCol >= 0 && newCol < grid[0].length &&
                    !visited[newRow][newCol] && grid[newRow][newCol] != 2 /* Wall */) {
                    
                    // Found a path from start to this cell
                    if (grid[newRow][newCol] == 0 /* Path */){
                        result.add(grid[newRow][newCol]);
                    }
                    
                    // If there's no wall, mark this cell as visited
                    visited[newRow][newCol] = true;

                    // Add neighboring cells to the queue
                    queue.offer(new int[]{newRow, newCol});
                }
            }
        }

        return result;
    }
}
