#include <iostream>
#include <vector>

// Function to transpose a matrix using cache-tiled approach.
std::vector<std::vector<int>> transposeMatrixCpu(const std::vector<std::vector<int>>& A) {
    const int rows = A.size();
    const int cols = A[0].size();

    // Tile size for caching
    const int tile_size = 32;

    std::vector<std::vector<int>> transposed(rows, std::vector<int>(cols));

    #pragma omp parallel for schedule(static)
    for (int i = 0; i < rows; ++i) {
        for (int j = 0; j < cols; ++j) {
            transposed[j][i] = A[i][j];
        }
    }

    return transposed;
}

// Main function to test the transposeMatrixCpu function.
int main() {
    std::vector<std::vector<int>> matrix = {{1, 2, 3}, {4, 5, 6}, {7, 8, 9}};

    auto start_time = std::chrono::high_resolution_clock::now();
    auto transposed_matrix = transposeMatrixCpu(matrix);
    auto end_time = std::chrono::high_resolution_clock::now();

    std::cout << "CPU Transpose Time: ";
    auto duration = std::chrono::duration_cast<std::chrono::microseconds>(end_time - start_time).count();
    std::cout << duration << " microseconds\n";

    return 0;
}
