```python
import sqlite3

class JobQueue:
    def __init__(self):
        self.conn = sqlite3.connect('job_queue.db')
        self.cursor = self.conn.cursor()
        
        # Create tables if they don't exist
        self.create_table()

    def create_table(self):
        sql = '''
            CREATE TABLE IF NOT EXISTS jobs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                state TEXT NOT NULL DEFAULT 'pending',
                created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
                data BLOB
            )
        '''
        self.cursor.execute(sql)
        self.conn.commit()
    
    def enqueue_job(self, data):
        try:
            self.cursor.execute("INSERT INTO jobs (data) VALUES (?)", (data,))
            self.conn.commit()
            return True
        except Exception as e:
            print(f"Error enqueuing job: {e}")
            return False
    
    def get_jobs(self, status=None):
        filters = []
        if status:
            filters.append(f"(state = '{status}')")
        
        sql = "SELECT * FROM jobs WHERE " + " AND ".join(filters)
        self.cursor.execute(sql)
        rows = self.cursor.fetchall()
        return [dict(row) for row in rows]
    
    def dequeue_job(self):
        try:
            self.cursor.execute("DELETE FROM jobs LIMIT 1")
            self.conn.commit()
            return True
        except Exception as e:
            print(f"Error dequeuing job: {e}")
            return False
    
    def update_job_status(self, job_id, new_state):
        sql = "UPDATE jobs SET state=? WHERE id=?"
        self.cursor.execute(sql, (new_state, job_id))
        self.conn.commit()
    
    def close_connection(self):
        self.conn.close()

# Example usage
queue = JobQueue()
queue.enqueue_job(b'{"task": "process_data"}')

print(queue.get_jobs(status='running'))
queue.update_job_status(1, 'completed')
print(queue.get_jobs())
```
