```python
import numpy as np

class SparseMatrix:
    def __init__(self):
        self.data = {}
    
    def insert(self, i, j, value):
        if (i, j) in self.data:
            self.data[(i, j)] += value
        else:
            self.data[(i, j)] = value
    
    def multiply_vector(self, vector):
        result = 0
        for key in self.data.keys():
            row, col = key
            result += self.data[key] * vector[col]
        return result

# Example usage
sparse_matrix = SparseMatrix()
data_points = [(1, 2, 3), (2, 4, -5), (3, 4, 6)]
vector = [1, 2]

for point in data_points:
    sparse_matrix.insert(*point)

result = sparse_matrix.multiply_vector(vector)
print(result)  # Output should be 7
```
