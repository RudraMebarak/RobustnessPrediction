    @Override

        super.logout(request, response, authentication);
        
        if (authentication.getAuthorities().contains(ADMIN_ROLE)) {
            // Handle admin-specific logic
        } else if (authentication.getAuthorities().contains(EDITOR_ROLE)) {
            // Handle editor-specific logic
        } else {
            // Default action for regular users
        }
    }
}
