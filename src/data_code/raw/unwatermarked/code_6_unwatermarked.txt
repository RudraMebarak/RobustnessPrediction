def lsd_radix_sort(arr):
    max_val = max(arr)
    
    # Perform counting sort for each digit position from right to left
    for exp in range(1, 33):  # For 32-bit unsigned integer
        count = [0] * (max_val // exp + 1)
        
        for num in arr:
            index = (num // exp) % 10
            count[index] += 1
        
        for i in range(1, len(count)):
            count[i] += count[i - 1]
        
        output = [0] * len(arr)
        
        for num in reversed(arr):
            index = (num // exp) % 10
            output[count[index] - 1] = num
            count[index] -= 1
        
        arr.clear()
        arr.extend(output)

# Example usage:
arr = [9, 8, 7, 6, 5, 4, 3, 2, 1, 0]
lsd_radix_sort(arr)
print("Sorted array:", arr)
