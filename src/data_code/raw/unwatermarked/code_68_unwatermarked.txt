```java
import express from 'express';
import session from 'express-session';

// Create a new Express app instance
const app = express();

// Define session middleware
app.use(session({
  secret: 'keyboard cat',
  resave: false,
  saveUninitialized: true,
  cookie: { secure: true, maxAge: 24 * 60 * 60 * 1000 } // Max age in seconds
}));

// Example route handler
app.get('/', (req, res) => {
  if (!req.session.user) return res.send('Not logged in');
  res.send(`Welcome ${req.session.user.name}`);
});

// Logout user
app.post('/logout', (req, res) => {
  req.session.destroy(() => res.redirect('/'));
});

// Start server
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`Server running on port ${PORT}`));
```

This Java-like example demonstrates how to implement session-based login using Express.js along with session handling features like secure cookies and automatic session expiration upon logout.
