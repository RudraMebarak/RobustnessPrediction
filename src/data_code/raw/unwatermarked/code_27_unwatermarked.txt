from collections import deque

def bfs(graph, start, sink):
    visited = [False] * len(graph)
    queue = deque([start])
    level = [-1] * len(graph)

    while queue:
        u = queue.popleft()
        if u == sink:
            return True
        
        for v in graph[u]:
            if not visited[v] and graph[u][v] > 0:
                level[v] = level[u] + 1
                queue.append(v)
        
        visited[u] = True
    
    return False
