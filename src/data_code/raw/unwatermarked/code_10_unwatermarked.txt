import java.util.*;

class SensorReading implements Comparable<SensorReading> {
    int timestamp;
    double reading;

    public SensorReading(int timestamp, double reading) {
        this.timestamp = timestamp;
        this.reading = reading;
    }

    @Override
    public int compareTo(SensorReading other) {
        return Double.compare(this.reading, other.reading);
    }
}

public class LazyMerge {

    private PriorityQueue<SensorReading> heap = new PriorityQueue<>();

    // Merge function to add all sensor readings into the heap.
    void addAll(Iterator<? extends SensorReading> iterator) {
        while (iterator.hasNext()) {
            heap.add(iterator.next());
        }
    }

    // Merge function to retrieve and remove the minimum reading from the heap.
    SensorReading next() throws NoSuchElementException {
        if (heap.isEmpty())
            throw new NoSuchElementException("No more readings available");
        
        SensorReading result = heap.poll();
        if (!heap.isEmpty())
            heap.offer(heap.poll());
        return result;
    }

    public static void main(String[] args) {
        List<ScannerReading> readings = Arrays.asList(new SensorReading(100, 2), new SensorReading(150, 3), new SensorReading(200, 4));
        LazyMerge merger = new LazyMerge();

        // Add all readings to the heap
        merger.addAll(readings.iterator());

        try {
            while(true) {
                System.out.println(merger.next().reading); // Prints the minimum reading
            }
        } catch (NoSuchElementException e) {
            System.out.println(e.getMessage()); // Prints "No more readings available"
        }
    }
}

This Java program defines a `SensorReading` class that stores timestamps and readings as double values. It also includes a `LazyMerge` class with methods for merging sensor readings from an iterable collection and retrieving the minimum reading one at a time. The `main` method demonstrates how to use these classes to merge multiple sorted sensor readings.
