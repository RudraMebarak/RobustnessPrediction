from collections import defaultdict

def bfs(graph, start):
    visited = set()
    queue = [start]
    
    while queue:
        node = queue.pop(0)
        
        if node not in visited:
            visited.add(node)
            
            # Get adjacent nodes with their distances from start
            neighbors_with_distance = []
            for neighbor, distance in graph[node].items():
                if neighbor not in visited:
                    neighbors_with_distance.append((neighbor, distance))
            
            # Sort by distance (ascending) then by alphabetical order of node names
            sorted_neighbors_with_distance = sorted(neighbors_with_distance, key=lambda x: (x[1], x[0]))
            
            # Add unvisited neighbors to the queue
            for neighbor, _ in sorted_neighbors_with_distance:
                if neighbor not in visited:
                    queue.append(neighbor)

# Example usage:
graph = {

    'B': {'A': 1, 'D': 3},

bfs(graph, 'A')
