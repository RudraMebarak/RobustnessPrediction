  const [selectedFile, setSelectedFile] = useState(null);
  const [foldersExpanded, setFoldersExpanded] = useState({

    currentFolder: false

  const handleExpand = (folderName) => {
    if (folderName === "root") return;
    setFoldersExpanded((prevState) => ({
      ...prevState,
      [folderName]: !prevState[folderName]
    }));
  };

  const handleSelect = (fileOrFolder) => {
    setSelectedFile(fileOrFolder);
  };

    <div key={item.name} className={`item ${depth * 2 + 1}`}>
      {item.isFile ? (
        <span>{item.name}</span>

        <>
          <button onClick={() => handleExpand(item.name)}>+</button>
          <div className="sub-item">
            <div onClick={() => handleSelect(item)}>
              {item.name}
            </div>
          </div>
        </>
      )}
    </div>

    if (hasChildren || (!isRoot && !foldersExpanded[lastFolder])) {
