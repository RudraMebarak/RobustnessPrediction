#include <iostream>
#include <fstream>
#include <filesystem>

// Function to compare two directories recursively.
void compare_directories(const std::string& dir1, const std::string& dir2) {
    // Check if both paths are valid directories.
    if (!std::filesystem::exists(dir1) || !std::filesystem::exists(dir2)) {
        throw std::runtime_error("One of the provided paths does not exist.");
    }

    // Compare each item in the first directory with its corresponding item in the second directory.
    for (const auto& entry : std::filesystem::directory_iterator(dir1)) {
        const auto path1 = entry.path();
        const auto path2 = dir2 / path1.filename();

        // If an entry exists in one but not the other, copy it from the first to the second.
        if (!std::filesystem::exists(path2)) {
            std::cout << "Copying " << path1 << " to " << path2 << std::endl;
            try {
                std::filesystem::copy_file(entry.path(), path2);
            } catch (const std::exception& e) {
                std::cerr << "Error copying " << path1 << ": " << e.what() << std::endl;
            }
        }
    }
}

int main() {
    // Example usage:
    compare_directories("/path/to/source/directory", "/path/to/destination/directory");
    
    return 0;
}
