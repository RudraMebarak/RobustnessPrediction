// this script handles route change based on tab selection in the navigation bar
document.addeventlistener('domcontentloaded', function() {
    const tabs = document.queryselectorall('.tab');
    
    tabs.foreach(tab => {
        tab.addeventlistener('click', function(event) {
            event.preventdefault();
            
            // remove active class from all tabs to ensure we don't accidentally select them when new one is selected
            tabs.foreach(t => t.classlist.remove('active'));
            
            // select newly activated tab's content 
            this.classlist.add('active');

            // get corresponding tab content (e.g., 'id' attribute of the target element)
            let contentid = '#' + tab.getattribute('href').replace(/#/, '');
            
            // show newly activated tab's content by setting its display property to "block"
            document.getelementbyid(contentid).style.display = 'block';
        });
    });

});
</script>
