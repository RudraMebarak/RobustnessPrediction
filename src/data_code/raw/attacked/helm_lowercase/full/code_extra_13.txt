```jsx
import react from 'react';
import moment from 'moment';

class calendar extends react.component {
  state = {
    year: new date().getfullyear(),
    monthindex: new date().getmonth(), // current month index (0-based)
    selecteddate: new date()
  };

  handlepreviousmonth = () => {
    this.setstate((state) => ({
      monthindex: state.monthindex - 1 >= 0 ? state.monthindex - 1 : 11,
      selecteddate: new date(state.year, state.monthindex - 1)
    }));
  };

  handlenextmonth = () => {
    this.setstate((state) => ({
      monthindex: state.monthindex + 1 <= 11 ? state.monthindex + 1 : 0,
      selecteddate: new date(state.year, state.monthindex)
    }));
  };

  render() {
    const { year, monthindex } = this.state;

    return (
      <div>
        <h2>{new date(year, monthindex).tolocalestring('default', { month: 'short' })}</h2>
        <button onclick={this.handlepreviousmonth}>previous month</button>
        <button onclick={this.handlenextmonth}>next month</button>

        {/* months are rendered in an unordered list */}
        <ul>
          {array.from({ length: 6 }, (_, i) => i % 2 === 0 ? null : (<li key={i} classname="month-item">
            <p>{new date(year, monthindex + (i / 2), 1).tolocaledatestring('default', { day: 'numeric' })}
              {' - '}
              {new date(year, monthindex + (i / 2), new date(year, monthindex + (i / 2), 32)).tolocaledatestring('default', { day: 'numeric' })}
            </p>
          </li>))}
        </ul>
      </div>
    );
  }
}

export default calendar;
```

this react component `calendar` creates a simple monthly grid where you can select dates with event markers. it utilizes `usestate` for managing the current date's year, month index, and selected date. the user can navigate between months through buttons, and the grid is dynamically rendered based on the current view month's days. each row in the grid represents two weeks of days, with even indices being blank to represent gaps between full months.
