import react, { usestate } from 'react';

// define custom hook to handle selection change event for each item
const useselectionchange = (items) => {
  const [selectedindex, setselectedindex] = usestate(null);

  return ({ key }) => {
    if (key === 'select') {
      setselectedindex(items.findindex(item => item.value === items[key].value));
    }
    return selectedindex;
  };
};

function multiselectdropdown() {
  // mock data for demonstration purposes
  const mockdata = [
    { value: 'option 1', label: 'option 1' },
    { value: 'option 2', label: 'option 2' },
    { value: 'option 3', label: 'option 3' }
  ];

  const selectitems = ['option 1', ...mockdata.map(item => item.value)];

  return (
    <div>
      <input type="hidden" id="selected-chip" value={null} />
      <ul role="listbox">
        {selectitems.map((item, i) => (
          <li key={i} ref={(node) => (chipinputref.current = node)} classname="selected-chip">
            <span>{item}</span>
            <button type="submit">x</button> {/* placeholder for "remove" button */}
          </li>
        ))}
      </ul>
      <label htmlfor="select-chip">{mockdata[0].label}</label>
      <script defer dangerouslysetinnerhtml={{ __html: `window.addeventlistener('keydown', onkeydown);` }}></script>
      <script dangerouslysetinnerhtml={{__html='window.onselectionchange = onchange;'}}></script>

      <script defer dangerouslysetinnerhtml={{
        __html:
          `
              window.onselectionchange = function(e){
                var chipelm = document.getelementbyid("selected-chip");
                var index = chipelm.getattribute("aria-setsize") - 1;
                chipelm.setattribute("aria-selected", e.type !== 'blur');
                if (e.type === 'keyup') {
                    console.log(index)
                    window.onkeydown();
                }
              }
              
              window.onchange = function (index) {
                let newvalue = "";
                for (var i = 0; i <= index; i++) {
                  newvalue += selectitems[i];
                }
                window.onselect(newvalue);
              }
          
              window.onkeydown = () => {
                document.queryselector("#select-chip").click();  
              };
          `
      }}></script>
    </div>
  );
}

export default multiselectdropdown;
