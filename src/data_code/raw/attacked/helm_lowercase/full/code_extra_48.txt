    synchronizationutility(const std::string& local_dir, const std::string& remote_dir)

        std::filesystem::directory_iterator localit(localdir);
        std::filesystem::directory_iterator remoteit(remotedir);

            if (!std::filesystem::exists(*localit)) { // file is missing locally but exists remotely
                copyfilestoremote(*localit);                
            }
