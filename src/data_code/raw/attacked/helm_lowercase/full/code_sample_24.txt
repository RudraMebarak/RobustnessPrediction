import java.util.*;

public class mazebfs {
    static final int[][] dirs = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};

    public list<list<integer>> bfs(int[][] grid) {
        boolean[][] visited = new boolean[grid.length][grid[0].length];
        queue<int[]> queue = new arraydeque<>();
        
        // initialize starting position and distance
        queue.offer(new int[]{0, 0});
        visited[0][0] = true;
        int distance = 1;

        list<list<integer>> path = new arraylist<>();
        list<list<integer>> paths = new arraylist<>();
        
        while (!queue.isempty()) {
            list<integer> pathelement;
            
            if (visited[grid.length - 1][grid[0].length - 1]) break; // exit condition
            
            list<integer> temppath = new arraylist<>();
            for (int[] element : queue) {
                temppath.add(element[1]);
            }
            
            path.add(temppath);
            queue.clear();
            
            for (int[] dir : dirs) {
                int newx = element[0] + dir[0], newy = element[1] + dir[1];

                if (newx >= 0 && newx < grid.length && newy >= 0 && newy < grid[0].length && !visited[newx][newy]) {
                    queue.offer(new int[]{newx, newy});
                    visited[newx][newy] = true;
                    distance += 1;
                }
            }

            // reconstruct the path from end point back to start point
            list<integer> path2 = new arraylist<>();
            int[] currpos = grid[grid.length - 1][grid[0].length - 1];
            while (currpos[0] != 0 || currpos[1] != 0) {
                list<integer> temppath = new arraylist<>();
                temppath.add(currpos[0]);
                temppath.add(currpos[1]);

                path2.add(0, temppath); // add in reversed order to maintain original direction

                int[] prevpos = grid[currpos[0]][currpos[1]];
                currpos = prevpos;
            }
            collections.reverse(path2);

            // append reconstructed path at current node's depth level
            for (list<integer> newpath : path2)
                path.add(0, newpath);

        }

        return path;
    }
