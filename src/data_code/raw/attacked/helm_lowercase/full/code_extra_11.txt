const folder = (props) => {
  const [isexpanded, setexpanded] = react.usestate(false);
  let childrentodisplay;

  if (isexpanded) {
    // determine what child elements should be displayed based on folder's content
    if (props.folder.files.length > 0) {
      childrentodisplay = props.folder.files;
    }
    if (props.folder.directories.length > 0) {
      childrentodisplay = props.folder.directories;
    }
  }

  return (
    <div onclick={() => setexpanded(!isexpanded)}>
      {childrentodisplay && <ul>
        {childrentodisplay.map((item, index) => (
          <li key={index}>
            {item.type === 'file' ? (<a href={item.url}>{item.name}</a>) :
              <folder {...{ folder: item, ...props }} />}
          </li>)}
        </ul>}
      
      {/* breadcrumb navigation */}
      <breadcrumb />

      {/* empty folder icon */}
      {(props.folder.files.length <= 0 && !props.folder.directories.length) &&
        <i classname="fas fa-times-circle" title='no contents'></i>}
      
      {/* default folder icon */}
      {(props.folder.files.length <= 0 && !props.folder.directories.length) &&
        <i classname="fas fa-folder"></i>}

      {/* expand icon */}
      {(props.folder.files.length <= 0 && !props.folder.directories.length) &&
        <i classname="fas fa-caret-up" title='show contents...'></i>}

      {(props.folder.files.length <= 0 && !props.folder.directories.length) &&
        <i classname="fas fa-caret-down" title='hide contents...'></i>}
      
    </div>)
};
