import react, { usestate } from 'react';

// define the expenseitem component to display transaction details
const expenseitem = ({ id, description, amount }) => (

  </div>

// define the categories component to show category names
const categories = ({ categories }) => {
  return (
    <div>
      {object.keys(categories).map((category) => (
        <span key={category} classname={`category ${categories[category].active ? "active" : ""}`}>
          {category}
        </span>
      ))}
    </div>
  );
};

// define the addexpenseform component to handle expense addition
const addexpenseform = ({ onadd }) => (
  <form onsubmit={(e) => e.preventdefault()}>
    <label htmlfor="description">description:</label>
    <input type="text" id="description" name="description"></input>
    <label htmlfor="amount">amount (usd):</label>
    <input type="number" id="amount" name="amount"></input>
    <button type="submit">add expense</button>
  </form>

  <ul>

);

// define the monthlytotals component to display summary of each month's total expenditures
const monthlytotals = ({ monthsdata }) => {
  const databymonth = object.values(monthsdata);
  const formatteddata = databymonth.map((data) => ({
    ...data,
    year: new date(data.date).getfullyear(),
  }));

  const monthsarray = ['january', 'february', 'march', 'april', 'may', 'june', 'july', 'august', 'september', 'october', 'november', 'december'];
  const groupedmonths = {};

  monthsarray.foreach((month) => {
    let sum = 0;
    formatteddata.foreach((item) => {
      if (item.month === month.tolowercase()) {
        sum += item.total || 0;
      }
    });

    groupedmonths[month] = sum > 0 ? sum.tolocalestring('en-us', { style: 'currency', currency: 'usd' }) : '--';
  });

  return (
    <>
      <h2>total expenditures by month:</h2>
      {object.entries(groupedmonths).map(([month, total]) => (
        <div key={month}>
          <strong>{month}: </strong> {total}
        </div>
      ))}
    </>
  );
};
