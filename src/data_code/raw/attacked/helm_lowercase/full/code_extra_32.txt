```html
<!doctype html>
<html lang="en">
<head>
    <meta charset="utf-8">
    <title>recipe collection</title>
    <style>
        body { font-family: arial, sans-serif; }
        header { background-color: #f1f1f1; padding: 20px; text-align: center; }
        .categories { display: flex; justify-content: space-around; }
        .category { width: 35%; margin-bottom: 20px; }
        .category h2 { color: #333; }
        .recipes-container ul li { list-style-type: none; padding-left: 20px; }
        .recipes-container ul li a { text-decoration: none; color: #007bff; }
        .recipes-container ul li a:hover { text-decoration: underline; }
        footer { text-align: center; padding-top: 20px; }
    </style>
</head>
<body>

<header>
    <h1>recipe collection</h1>
</header>

<div class="categories">
    <div class="category" onclick="togglecategory('categorize_all')">all recipes</div>
    <div class="category" id="vegetarian" onclick="togglecategory('vegetarian')">vegetarian</div>
    <div class="category" id="glutenfree" onclick="togglecategory('glutenfree')">gluten free</div>
</div>

<div class="recipes-container" style="display:none;">
    <ul>
        <!-- recipe items will be dynamically inserted here -->
    </ul>
</div>

<script>
function togglecategory(category) {
    const elements = document.queryselectorall('.recipes-container ul li');
    elements.foreach(item => item.style.display = 'none');

    if (category === "all") {
        return;
    }

    document.getelementbyid(category).style.display = 'block';
}

document.getelementbyid('vegetarian').addeventlistener('click', function() {
    togglecategory('vegetarian');
});
document.getelementbyid('glutenfree').addeventlistener('click', function() {
    togglecategory('glutenfree');
});

// add event listeners to trigger clicks on categories
const categories = document.getelementsbyclassname('categories div')[0].getelementsbytagname('div');
for (let i = 1; i <= categories.length; i++) {
    const catbutton = document.getelementbyid(`category${i}`);
    catbutton.addeventlistener('click', () => {
        togglecategory(`category${i - 1}`);
    });
}
</script>

<footer>
    copyright © 2023
</footer>

</body>
</html>
```
