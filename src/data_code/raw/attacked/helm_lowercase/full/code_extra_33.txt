<script>
// example data (replace this with actual data from your db)
const products = [
{id: 101, name: "widget", qty: 50, price: 10},
{id: 102, name: "gadget", qty: 30, price: 20}
];

function populatetable() {
  const tbody = document.getelementbyid("inventorydata");
  tbody.innerhtml = ''; // clear previous contents
  
  products.foreach(product => {
    let row = '<tr>';
    row += `<td>${product.id}</td>`;
    row += `<td>${product.name}</td>`;
    row += `<td>${product.qty}</td>`;
    row += `<td>$${product.price.tofixed(2)}</td>`;
    row += '<td><a href="#">edit/delete</a></td>';
    row += '</tr>';
    tbody.appendchild(document.createelement('tr')).innerhtml = row; // append each row
  });
}

document.addeventlistener('domcontentloaded', () => {
populatetable(); // initial load

document.getelementbyid('category').addeventlistener('change', e => {
  let selectedcategory = e.target.value;

  if(selectedcategory === '') {
    populatetable(); // reload initial state when category is cleared
  } else {
    populatetable(); // filter after clicking category change event
  }
});

});
</script>

</body>
