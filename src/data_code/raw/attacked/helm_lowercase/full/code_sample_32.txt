// buffer size in bytes (1 byte per integer)
#define buffer_size 1024

// circular buffer structure
struct buffer {
    int* buffer;
    int front;
    int rear;
    int capacity;
};

void initializebuffer(struct buffer* buffer) {
    buffer->buffer = (int*)malloc(buffer_size * sizeof(int));
    if (buffer->buffer == null) {
        fprintf(stderr, "memory allocation failed\n");
        exit(exit_failure);
    }
    buffer->front = 0; // position of the first available element
    buffer->rear = 0; // position of the last used element before being placed back on stack
    buffer->capacity = buffer_size; // maximum number of elements in the buffer
}

void pushtobuffer(struct buffer* buffer, int item) {
    pthread_mutex_lock(&mutex); // acquire lock to ensure thread safety

    pthread_cond_signal(&notempty); // notify waiting consumer threads
    pthread_mutex_unlock(&mutex); // release lock after operation completed safely
}
