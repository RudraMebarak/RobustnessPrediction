import numpy as np

def bfs(graph, start_node, sink):
    """
    perform a breadth-first search (bfs) to find an augmenting path from source to sink in residual graph.
    
    parameters:
        graph: 2d list of lists representing the residual graph after applying all edge-disjoint paths found so far.
        start_node: the source node for bfs.
        sink: the target node for bfs.
        
    returns:
        found path through residual graph; false if no such path exists.
    """
    queue = [start_node]
    visited = {node: false for row in range(len(graph)) for node in row}
    
    while queue:
        current_node = queue.pop(0)
        if current_node == sink:
            return true
        
        for neighbor_index, weight in enumerate(graph[current_node]):
            if weight > 0 and not visited[neighbor_index]:
                queue.append(neighbor_index)
                visited[neighbor_index] = true
    
    return false

def get_flow_remaining(graph, flow_dict, vertex1, vertex2):
    """
    update flow_dict with remaining capacities on edges between vertex1 and vertex2.

    parameters:
        graph: 2d list of lists representing the residual graph.
        flow_dict: dictionary mapping each node pair to its incoming flow.
        vertex1: source node.
        vertex2: target node.
    """
    capacity = graph[vertex1][vertex2]['capacity']
    new_flow = min(flow_dict[(vertex1, vertex2)], capacity)
    capacity_after_flows = capacity - new_flow
    graph[vertex1][vertex2]['flow'] += new_flow
    graph[vertex2][vertex1]['flow'] -= new_flow

    return capacity_after_flows
