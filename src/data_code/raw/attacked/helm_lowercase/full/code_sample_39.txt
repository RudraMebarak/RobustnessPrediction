```c
#include <sys/mman.h>
#include <fcntl.h>
#include <unistd.h>
#include <stdio.h>

void handle_errors() {
    perror("mmap");
    exit(exit_failure);
}

int main(int argc, char* argv[]) {
    if (argc != 2) {
        fprintf(stderr, "usage: %s filename\n", argv[0]);
        return exit_failure;
    }

    int fd = open(argv[1], o_rdonly);
    if (fd == -1)
        handle_errors();

    off_t size = lseek(fd, seek_end, seek_set);
    close(fd);

    if (size < 0) {
        handle_errors();
    }
    
    long num_bytes_mapped = size;

    void* addr;
    int fd2;
    struct stat buf;
    fd2 = creat("/dev/zero", s_iread|s_iwrite);
    if (fd2 == -1)
        handle_errors();
    if (fstat(fd2, &buf) == -1)
        handle_errors();
    if (mmap(null, buf.st_size, prot_read | prot_write, map_shared, fd2, 0) == map_failed) {
        handle_errors();
    }
    addr = mmap(null, size, prot_read | prot_write, map_private, fd, 0);
    if (addr == map_failed) {
        handle_errors();
    }
    
    int nnewlines = 0;
    off_t newl_offset = 0;
    while (newl_offset < size) {
        char ch = getc(fd2);
        if (ch != eof && ch == '\n') {
            ++nnewlines;
        }
        
        // seek forward by one byte for next iteration
        seek(fd2, newl_offset + 1, seek_cur);
        newl_offset += sizeof(long) << 3; // size of an unsigned long in bytes.
    }
    
    munmap(addr, num_bytes_mapped);
    close(fd2);
    unlink(argv[1]);

    printf("%ld newlines found.\n", nnewlines);
}
```
