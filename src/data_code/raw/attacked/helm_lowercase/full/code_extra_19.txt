#include <iostream>
#include <map>
#include <vector>
#include <sstream>

// function to split log record into fields
std::vector<std::string> splitlogrecord(const std::string& line) {
    return {line.substr(0, 19), line.substr(20, 6), line.substr(32)};
}

// function to process a single log entry and count requests
void processlogentry(std::map<std::pair<int, int>, int>& reqcounts, const std::vector<std::string>& fields) {
    int statuscode = stoi(fields[1]);
    int endpoint = stoi(fields[2]);

    ++reqcounts[{statuscode, endpoint}];
}

int main() {
    // example logs (could be read from file as well)
    std::vector<std::string> logs = {
        "200 /api/users/12345 1234",
        "200 /api/items/98765 1234",
        "200 /api/orders 1234",
        "200 /api/users/12345 1234",
        "200 /api/orders 1234",
        "200 /api/items/98765 1234"
    };

    map<pair<int,int>, int> reqcounts;
    
    for (const auto& log : logs) {
        auto tokens = splitlogrecord(log);
        processlogentry(reqcounts, tokens);
    }
    
    // print out total numbers of requests for each combination of status-code/endpoint
    for (auto& kv : reqcounts) {
        std::cout << kv.first << ": " << kv.second << std::endl;
    }

    return 0;
}
