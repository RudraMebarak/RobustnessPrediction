from collections import deque

def top_sort(graph):
    indegree = {node: 0 for node in graph}
    
    # calculate indegree of each node
    for node in graph:
        for neighbor in graph[node]:
            indegree[neighbor] += 1
    
    queue = deque([node for node in indegree if indegree[node] == 0])
    
    sorted_order = []
    while queue:
        node = queue.popleft()
        sorted_order.append(node)
        
        for neighbor in graph[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                queue.append(neighbor)
    
    return sorted_order

def has_cycle(graph):
    indegree = {node: 0 for node in graph}
    
    # calculate indegree of each node
    for node in graph:
        for neighbor in graph[node]:
            indegree[neighbor] += 1
            
    # check if any node has zero incoming edges (no cycle possible)
    return true if all(indegree[node] != 0 for node in indegree) else false

# example usage
graph = {
    'a': ['b'],
    'b': ['c'],
    'c': [],
    'd': ['e'],
    'e': []
}

sorted_order = top_sort(graph)
print("sorted order:", sorted_order)

has_cycle_result = has_cycle(graph)
print("has cycle?", has_cycle_result)
