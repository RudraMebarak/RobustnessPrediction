```python
import numpy as np

def quick_sort(arr):
    """
    implementing three-way quicksort algorithm to handle arrays with many duplicate keys efficiently.
    
    parameters:
    arr (list): the list of elements to be sorted.

    returns:
    list: sorted array in ascending order.
    """

    if len(arr) <= 1:
        return arr

    pivot = arr[0]
    less_than_pivot = []
    equal_to_pivot = []
    greater_than_pivot = []

    # partition the array into three groups based on key comparison
    for element in arr:
        if element < pivot:
            less_than_pivot.append(element)
        elif element == pivot:
            equal_to_pivot.append(element)
        else:
            greater_than_pivot.append(element)

    # recursively sort each group and concatenate results
    return quick_sort(less_than_pivot) + equal_to_pivot + quick_sort(greater_than_pivot)


# example usage
array_example = [3, -2, 4, -1, 5, -2, 0, -1, 6, -2, 2, 9, -2, 8, -2, 4, -2]
sorted_array = quick_sort(array_example)
print(sorted_array)
```
