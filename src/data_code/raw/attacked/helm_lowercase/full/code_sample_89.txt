# define a class for representing a token (number, operator)
class token:
    def __init__(self, type, value):
        self.type = type
        self.value = value

    def __repr__(self):
        return f"token({self.type}, {self.value})"

# define a stack to hold tokens in order
class stack:
    def __init__(self):
        self.items = []

    def push(self, item):
        self.items.append(item)

    def pop(self):
        if not self.is_empty():
            return self.items.pop()
        else:
            raise indexerror("pop from empty stack")

    def is_empty(self):
        return len(self.items) == 0

    def peek(self):
        if not self.is_empty():
            return self.items[-1]
        else:
            return none
