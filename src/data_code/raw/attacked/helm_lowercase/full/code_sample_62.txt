```python
import sqlite3

class batchcsvimporter:
    def __init__(self, database_path):
        self.conn = sqlite3.connect(database_path)
        self.cursor = self.conn.cursor()
    
    def import_data(self, file_path, chunk_size=100):
        with open(file_path) as file:
            line_count = 0
            while true:
                data_chunk = file.read(chunk_size)
                if not data_chunk:
                    break
                
                # validate each row in the chunk
                valid_rows = []
                invalid_rows = []
                
                try:
                    cursor = self._get_sql_cursor()  # assume this method connects to the appropriate table
                    query = "insert into your_table_name (col1, col2, ...) values (?, ?, ...)"
                    cursor.executemany(query, zip(*[iter(data_chunk.splitlines())] * len(cursor.description)))
                    valid_rows.extend([row[0] for row in cursor.rowcount])
                    
                except sqlite3.integrityerror:  # for validation errors
                    invalid_rows.append(line_count)
                    
                finally:
                    self._close_sql_connection()  # ensure connection is closed after processing
        
        print(f"processed {len(valid_rows)} out of {line_count} lines.")
        print(f"{sum(invalid_rows)} entries were rejected due to integrity violations.")

    def _get_sql_cursor(self):
        return self.cursor.execute('begin')  # replace 'your_table_name' with actual table name
        
    def _close_sql_connection(self):
        self.cursor.close()
        self.conn.commit()
        self.conn.close()

# example usage
if __name__ == "__main__":
    db_path = 'path/to/your/database.db'
    csv_path = 'path/to/your/csv.csv'
    importer = batchcsvimporter(db_path)
    importer.import_data(csv_path, chunk_size=100)
```

note: the `_get_sql_cursor` function assumes you have already set up the sqlite connection and prepared an sql statement compatible with `executemany`. adjust these parts according to your specific setup and requirements.
