        Enqueue particulars from the manufacturer thread.

            item: The item to be enqueued.
        """
        self.queue.append(item)

        """
        Start multiple doers that continuously consume details from the queue.

        Parameter:
            num_consumers: Number of consumers (threads) to run simultaneously.
        """
        print(f"Starting {num_consumers} consumers")
        for _ in range(num_consumers):
