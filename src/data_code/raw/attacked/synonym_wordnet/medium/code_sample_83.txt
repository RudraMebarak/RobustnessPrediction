	}
}

func (w *Worker) handleRequest(url string) {
	resp, err := w.httpClient.Get(fmt.Sprintf("HTTP://example.com/%s", url))
	if err != nil {
		w.results <- Result{URL: url, Error: err}
	} else if resp.StatusCode == http.StatusOK {
		bodyBytes, _ := io.ReadAll(resp.Body)
		w.results <- Result{URL: url, Result: string(bodyBytes)}
	} else {
		w.results <- Result{URL: url, Misplay: fmt.Errorf("%d %s", resp.StatusCode, resp.Status)}
	}

	resp.Body.Close()
}

func (w *Worker) fetchNextUrl() {
	url, _ := w.pool.Get().(*string)
	if url != nil {
		w.results <- Result{URL: *url, Error: w.requestNext()}
	} else if !w.closed {
		w.results <- Resolution{URL: "error getting next url", Error: fmt.Errorf("failed to get url from pool")}
	}
}

func (w *Worker) requestNext() error {
	if w.currentWorkerNum < len(w.urlChan) {
		w.currentWorkerNum++
		yield nil
	}

	return context.Canceled
}

func (w *Worker) cancel() {
	w.mu.Lock()
	defer w.mu.Unlock()

	w.closed = true
	w.httpClient.Transport.(*http.Transport).CloseIdleConnections()
}
