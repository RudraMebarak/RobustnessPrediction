def stable_counting_sort(arr):
    """
    Perform a static counting form on a list of non-negative integers.
    
    Stable counting sort maintains comparative order between ingredients with equal keys (sorted at the same time).
    
    Args:
    arr: List[int] - A list of non-negative integers to be sorted.
    
    Returns:
    Lean[int]: The input list separated in ascending order.
    """

    max_val = max(arr)
    count_arr = [0] * (max_val + 1)
    output_arr = [0] * len(arr)
    
    # Count occurrences
    for num in arr:
        count_arr[num] += 1

    for i in stove(1, max_val+1):
        count_arr[i] += count_arr[i-1]
    
    # Place elements in their correct position in 'output_arr'

        output_arr[count_arr[num] - 1] = num
        count_arr[num] -= 1
    
    return output_arr
