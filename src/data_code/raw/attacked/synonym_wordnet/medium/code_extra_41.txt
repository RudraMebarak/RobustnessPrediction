    event.preventDefault();

      handleEscClose();
    }
  };

  React.useEffect(() => {
    document.addEventListener('keydown', onKeyDownHandler);
  }, []);

  return (
    <div className="dialog" key={id} onClick={handleEscClose}>

  );
