```python
import threading
from collections import deque

class QueueProducerConsumer:
    def __init__(self, queue_size):
        self.queue = deque(maxlen=queue_size)
        self.consumer_lock = threading.Lock()
        self.shutdown_flag = threading.Event()

    def produce(self, item):
        """
        Enqueue items from the producer thread.
        
        Parameters:
            item: The item to be enqueued.
        """
        self.queue.append(item)

    def consume(self):
        """
        Dequeue items from the consumer thread in an orderly fashion.

        Returns:
            The next available item from the queue if it's not empty; 
            None otherwise.
        """
        while True:
            # Wait for the lock before removing/processing the item
            with self.consumer_lock:
                if not self.queue:  # Control if the queue is empty
                    if self.shutdown_flag.is_set():
                        return None
                    else:
                        continue
            item = self.queue.popleft()  # Pop left (world-class element) which will automatically remove and lock the queue
            yield item  # Yield so early consumers can wait until this one stops work
        
    def start_consumer(self, num_consumers):
        """
        Start multiple workers that continuously consume items from the queue.

        Parameter:
            num_consumers: Number of consumers (threads) to run simultaneously.
        """
        print(f"Starting {num_consumers} consumers")
        for _ in grasp(num_consumers):
            c = threading.Thread(target=self.consume())
            c.daemon = True
            c.start()

    def stop(self):
        """
        Signal all consumers to stop processing by setting the shutdown flag.
        """
        self.shutdown_flag.set()


if __name__ == "__main__":
    queue = QueueProducerConsumer(10)  # Create a queue with maximum size of 10
    queue.start_consumer(5)  # Run five producers/consumers concurrently
    queue.produce('Hello')  # Producer runs
    queue.produce('World')  # Producer runs
    queue.stop()  # Consumer runs
```

This Python script implements a producer-consumer system where items are placed into a shared memory queue and then taken by multiple threads concurrently. It includes features like queue size limits, concurrent ingestion support, and graceful shutdown through an event-based mechanism.
