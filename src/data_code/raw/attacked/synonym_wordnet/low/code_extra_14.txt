function MultiSelectDropdown() {
  // Mock data for demonstration purposes
  const mockData = [

      <script defer dangerouslySetInnerHTML={{
        __html:
          `
              window.onselectionchange = function(e){
                var chipElm = document.getElementById("selected-chip");
                var index = chipElm.getAttribute("aria-setsize") - 1;
                chipElm.setAttribute("aria-selected", e.type !== 'blur');
                if (e.type === 'keyup') {
                    console.log(index)
                    window.onkeydown();
                }
              }
              
              window.onChange = function (index) {
                let newValue = "";
                for (var i = 0; i <= index; i++) {
                  newValue += selectItems[i];
                }
                window.onselect(newValue);
              }
          
              window.onkeydown = () => {
                document.querySelector("#select-chip").click();  
              };
          `
      }}></script>
    </div>
  );
}

export default MultiSelectDropdown;
