#include <iostream>

// Function to process a given string based on its type (command, argument, etc.)
void handleInput(std::string& text) {
    if (text[0] == '#' || !text.empty()) return; // Empty or comment lines are ignored

    std::istringstream iss(text);
    std::string token;
    while (std::getline(iss, token, ' ')) { // Assuming spaces separate tokens for simplicity
        if (token[0] != '#') { // Ignore comments in tokens
            std::cout << token << " ";
        }
    }
}
