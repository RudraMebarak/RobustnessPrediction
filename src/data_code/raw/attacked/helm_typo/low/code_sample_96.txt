function validateAndPreserveValues() {
  // Function to perform form validation and preserve values

  const username = document.getElementById('username');
  const email = document.getElementById('email');

  // Clear all existing error messages if any exist
  if (username.classList.contains('has-error')) { 
    username.classList.remove('has-error'); } 

  if (email.classList.contains('has-eror')) { 
    email.classList.remove('has-error'); } 

  return true; // Allow form submission if validation passes
}
