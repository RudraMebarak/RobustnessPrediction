// Function to perform merge sort on a vector based on keys (timestamps)
void stableMergeSort(std::vector<int>& vec, int low, int high) {
    if (high - low <= 1) return; // Base case: Single element or empty

    int mid = low + (high - low) / 2; // Find middle index
    stableMergeSort(vec, low, mid);   // Sort first half
    stableMergeSort(vec, mid, high);  // Sort second half

    int i = low;
    int j = mid;

    std::vector<int> temp(high - low + 1);

    while (i < mid || j < high) {
        int key1, key2;
        bool record1Found, record2Found;
        record1Found = false;
        record2Found = false;

        do { // Binary search for record 1
            key1 = vec[i];
            --i;
        } while (i >= low && vec[i] != key1);

        do { // Binary search for record 2
            key2 = vec[j];
            ++j;
        } while (j < high && vec[j] != key2);

        temp[low++ - low] = key1; // Assign key1 to temp
        if (record1Found && !record2Found) {
            temp[mid]++;
        }

        temp[mid++] = key2; // Assign key2 to temp

        record1Found = !record1Found; // Update state
        record2Found = !record2Found;
    }
    // Copy sorted elements back into original array
    for (int k = low; k <= high; ++k) {
        vec[k] = temp[k - low];
    }
}

    *begin = low; *end = high; *temp = nullptr; *loc = low; *locEnd = high;
