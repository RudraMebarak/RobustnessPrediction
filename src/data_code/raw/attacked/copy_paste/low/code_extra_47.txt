// TokenBucketPolicyTokenBucket implements TokenBucketPolicy using a token bucket policy
class TokenBucketPolicyTokenBucket : public TokenBucketPolicy {
private:
    std::map<std::string, int> token_map;

public:
    void fill(Bucket* b) override {
        for (auto& kv : token_map) {
            if (b->tokens >= kv.second) break;
            else {
                --kv.second; // Remove tokens up to the available amount
                (*b).tokens += kv.second + kv.first; // Add tokens based on rate
                token_map.erase(kv);
                return;
            }
        }
        auto& kv = token_map.emplace(std::piecewise_construct,
                                     std::forward_as_tuple(b->name()),
                                     std::forward_as_tuple(1)).first;
        kv.second = kv.first; // Initial fill will increment this count
    }
};

// LeakyBucketPolicyLeakyBucket implements LeakyBucketPolicy using a leaky bucket policy
class LeakyBucketPolicyLeakyBucket : public LeakyBucketPolicy {
private:
    std::map<std::string, int> drain_times;

public:
    void drain(Bucket* b) override {
        drain_times[b->name()] += 1.0; // Increase draining counter each time
        if (drain_times[b->name()] > 3600) { // Assuming one hour as drain threshold
            drainer(); // Call draining function periodically
        }
    }

private:
    void drainer() {
        // Simulate draining process here
        for (auto& kv : drain_times) {
            kv.second -= kv.second / 2; // Simulate leaking out tokens over time
        }
    }
};
