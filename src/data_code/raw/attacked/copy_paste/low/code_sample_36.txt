#include <stdio.h>
#include <unistd.h>
#include <stdlib.h>

// Function to run commands in a pipeline.
int executePipeline(int num_args) {
    if (num_args < 1 || num_args > 5) {
        fprintf(stderr, "Invalid number of arguments for pipeline\n");
        return -1;
    }
    
    int child_pid[num_args];
    char **argv = (char**)malloc(num_args * sizeof(char*));
    
    // Allocate memory for arguments and prepare them
    argv[0] = NULL; // First argument is always an empty string to start the pipeline
    
    for (int i = 1; i <= num_args; ++i) {
        argv[i] = (char*)malloc(128 * sizeof(char)); // Allocate space based on potential argument length
        scanf("%s", argv[i]);
        
        if (i < num_args) {
            argv[i] += ' '; // Null terminate after each argument
        }
    }
    
    // Execute the pipeline
    for (int i = 1; i <= num_args; ++i) {
        int status;
        int pid = fork();
        
        if (pid == -1) {  // Error handling
            perror("Fork failed");
            exit(EXIT_FAILURE);
        } else if (pid != 0) {
            wait(&status); // Wait for parent process
        } else {
            execlp(argv[i], argv[i], (char *) NULL); // Execute the next argument as new process
        }
    }
    
    free(argv); // Free allocated memory
    
    return 0;
}

int main() {
    int num_args;
    
    printf("Enter numbers of args separated by space (max 5): ");
    scanf("%d", &num_args);
    
    if (executePipeline(num_args)) {
        fprintf(stderr, "Execution failed.\n");
        return -1;
    }
    
    return 0;
}
