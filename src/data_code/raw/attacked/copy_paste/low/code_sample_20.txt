import numpy as np

class SparseMatrix:
    def __init__(self):
        self.nonzero_elements = {}

    def insert(self, row, col, value):
        if (row, col) in self.nonzero_elements:
            self.nonzero_elements[(row, col)] += value
        else:
            self.nonzero_elements[(row, col)] = value

    def get_vector_product(self, vector):
        result = 0.0
        for (r, c), val in self.nonzero_elements.items():
            result += val * vector[c]
        return result
