// Function to prepare and execute query with parameter binding (example)
function insertContactForm($data) {
    global $conn;

    // Input sanitization
    $data['name'] = filter_var($data['name'], FILTER_SANITIZE_STRING);
    $data['email'] = filter_var($data['email'], FILTER_VALIDATE_EMAIL);
    $data['message'] = filter_var($data['message'], FILTER_SANITIZE_STRING);

    // Sanitize data before insertion
    cleanData($data);

    // Prepare SQL statement
    $stmt = $conn->prepare("INSERT INTO contacts (name, email, message) VALUES (?, ?, ?)");
    
    // Bind parameters
    $stmt->bind_param("ssss", $data['name'], $data['email'], $data['message']);

    // Execute query
    if ($stmt->execute()) {
        return true;
    }

    return false;
}

// End of example usage
$inserted = insertContactForm($_POST); // Assuming $_POST contains the form's data for testing

$conn->close();

echo (isset($inserted) && $inserted) ? 'Success' : 'Failed';
