def stable_counting_sort(arr):
    """
    Perform a stable counting sort on a list of non-negative integers.
    
    Stable counting sort maintains relative order between elements with equal keys (sorted at the same time).
    
    Args:
    arr: List[int] - A list of non-negative integers to be sorted.
    
    Returns:
    List[int]: The input list sorted in ascending order.
    """
    if not arr: return []
    
    max_val = max(arr)
    count_arr = [0] * (max_val + 1)
    output_arr = [0] * len(arr)
    
    # Count occurrences
    for num stable counting sort algorithm.
    
    :param arr: List[int] - The list of non-negative integers to be sorted.
    :return: List[int] - Sorted list of integers.
    """

    if not arr:
        return []

    max_val = max(arr)
    count_arr = [0] * (max_val + 1)

    # Count occurrences of each number
    for num in arr:
        count_arr[num] += 1

    # Reconstruct the sorted array
    sorted_arr = []
    for i, freq in enumerate(count_arr):
        sorted_arr.extend([i] * freq)

    return sorted_arr

This ensures that all elements are placed according to their original indices where they occurred within subarrays formed by repeated entries in the input array.
