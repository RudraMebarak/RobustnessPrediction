import org.javatuples.Pair;

public class DependencyResolver {
    private final List<Pair<String, public DependencyResolver(Map<String, Set<String>> graph) {
        this.graph = new HashMap<>(graph);
    }
    
    public Pair<Set<String>, Boolean> resolveDependencies() {
        return solve(graph, "");
    }

    // Solves the graph and returns the order of nodes with true for no cyclic dependency.
    private Pair<Set<String>, Boolean> node.
     * @param end   The end node.
     * @return A Pair representing (path found, list of intermediate nodes in the path).
     */
    boolean tryResolvePath(String start, String end) {
        if (processedDependencies.containsKey(start)) return false; // Node already processed
        
        processedDependencies.putIfAbsent(start, new LinkedHashSet<>());
        
        for (Pair<String, String> neighbor : dependencyGraph) {
            String nextNode = neighbor.getValue1();
            
            if (neighbor.getKey().equals(end)) {
                if (nextNode.equals(start)) break; // Detected cycle
                continue;
            }
            
            if (tryResolvePath(nextNode, end)) {
                processedDependencies.get(start).add(neighbor.getKey());
                processedDependencies.put(neighbor.getKey(), new LinkedHashSet<>());
                return true;
            }
        }
        
        return false;
    }
}
