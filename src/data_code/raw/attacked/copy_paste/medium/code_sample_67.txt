import hashlib

def create_hashed_password(password):
    # Unique salt for each hashed password
    salt = os.urandom(16)
    
    # Work factor is an additional parameter that defines how many rounds of hashing will take place.
    # For example, setting it to 20 would result in about 40 million iterations of SHA-256 hashing (20 rounds * 2^20).
    work_factor = 12
    
    # Generate a new hash based on the provided password, salt, and work factor
    hashed_password = hashlib.pbkdf2_hmac('sha256', password.encode('utf8'), salt, work_factor)
    
    return hashed_password + salt

# Example usage:
password = "mysecretpassword"
hashed = create_hashed_password(password)

print(f"Hashed Password: {hashed.hex()}") # The output will be a hexadecimal string representing the hashed password including the salt.

def verify_password(hashed_password, provided_password):
    try:
        # Compare the provided password with the stored hashed password
        if hashlib.pbkdf2_hmac('sha256', provided_password.encode(), bytes.fromhex(hashed_password[:32]), 100000) == \
                bytes.fromhex(hashed_password[32:]):
            print("Password is correct!")
            return True
        else:
            print("Incorrect password.")
            return False
    except Exception  # True if passwords match; False otherwise.
