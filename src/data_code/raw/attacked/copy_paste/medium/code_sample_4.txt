            int mid = tempData.size() / 2;
            List<Long> leftHalf = tempData.subList(0, mid);

    private static List<Long> merge(List<Long> leftHalf, List<Long> rightHalf) {
        List<Long> merged = new ArrayList<>();
        int i = 0, j = 0;

        while (i < leftHalf.size() && j < rightHalf.size()) {
            if (leftHalf.get(i) >= rightHalf.get(j)) {
                merged.add(rightHalf.get(j++));
            } else {
                merged.add(leftHalf.get(i++));
            }
        }

        while (i < leftHalf.size()) {
            merged.add(leftHalf.get(i++));
        }

        while (j < rightHalf.size()) {
            merged.add(rightHalf.get(j++));
        }

        return merged;
    }
