// Simulate AJAX call to fetch comparison results
async function getComparisonData() {
  const response = await fetch('https://api.insurance.com/compare?plan1=1&coverage1=A&value1=$50&exclusions1=123');
  if (!response.ok) throw new Error(`HTTP error! Status: ${response.status}`);
  return await response.json();
}

getComparisonData().then(data => {
  document.getElementById('result-list').innerHTML = data.map(resultItem).join('\n<li>' + resultItem + '</li>\n');

  return `<li>${item}</li>`;
