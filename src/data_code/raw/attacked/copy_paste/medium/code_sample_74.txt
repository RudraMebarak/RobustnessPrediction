def evaluate_imbalance(X_train, y_train, X_test, y_test, clf):
    """
    Evaluates the performance of a given imbalanced classifier on test data.

    Parameters:
        X_train: Training feature matrix of shape (n_samples_in, n_features).
        y_train: Training target variable vector of length n_samples_in.
        X_test: Test feature matrix of shape (n_samples_out, n_features).
        y_test: Test target variable vector of length n_samples_out.
        clf: The trained classifier to be evaluated.
        
    Returns:
        Balanced Accuracy score of the given imbalanced classifier on test data.
    """
    balanced_acc = []
    for _ in range(10):  # Repeat process multiple times for averaging
        calibrated_clf = CalibratedClassifierCV(base_estimator=clf,
                                                method='sigmoid',
                                                cv='prefit')
        calibrated_clf.fit(X_train, y_train)
        probas_pred = calibrated_clf.predict_proba(X_test)
        precision, recall, thresholds = precision_recall_curve(y_test, probas_pred[:, 1])
        average_precision = average_precision_score(y_test, probas_pred[:, 1])
        balanced_accuracy = (average_precision + average_precision) / 2 * max(precision, recall)
        balanced_acc.append(balanced_accuracy)
    return np.mean(np.array(balanced_acc))
