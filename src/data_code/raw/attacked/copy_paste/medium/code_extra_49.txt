// Function to create a TCP socket
int create_socket() {
    int sockfd;
    if ((sockfd = socket(AF_INET, SOCK_STREAM, 0)) == -1) {
        perror("socket creation failed");
        exit(EXIT_FAILURE);
    }
    return sockfd;
}

// Function to bind the socket to an IP address and port
void bind_socket(int& sock_fd, const char* host_ip, unsigned short port) {
    struct sockaddr_in serv_addr;
    memset(&serv_addr, '0', sizeof(serv_addr));

    serv_addr.sin_family = AF_INET;
    serv_addr.sin_port = htons(port);

    if (inet_pton(AF_INET, host_ip, &serv_addr.sin_addr) <= 0) {
        fprintf(stderr,"Invalid address/ Address not supported\n");
        exit(1);
    }

    if (bind(sock_fd, (struct sockaddr*)&serv_addr,sizeof(serv_addr)) == -1) {
        close(sock_fd);
        perror("bind() error");
        exit(1);
    }
}
