// Point lookup function to find an element with given key in the tree
int search(Node* root, int key) {
    if (root->is_leaf)
        return root->children[key].n != 0 ? key : -1;

    auto it = lower_bound(root->children.begin(), root->children.end(), key);
    
    if (*it == key)
        return *it;
    
    return search(static_cast<Node*>(*(--it)), key);
}
