import sqlite3
import os

class ExperimentRunner:
    def __init__(self):
        self.db_path = 'experiments.db'
        
        if not os.path.isfile(self.db_path):
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute('''CREATE TABLE IF NOT EXISTS end_time TIMESTAMP
            )
        ''')
        self.conn.commit()

    def add_experiment(self, config, seed, metrics=None):
        """Add a new experiment entry."""
        if metrics is None:
            metrics = {}
        
        data = {'config': str(config), 
                'seed': seed,
                'metrics': metrics}
        
        query = "INSERT INTO experiments (config, seed, metrics) VALUES (?, ?, ?)"
        self.c.execute(query, tuple(data.values()))
        self.conn.commit()

    def get_metrics(self, experiment_id):
        """Get metrics for a specific experiment."""
        query = "SELECT * FROM experiments WHERE id=?"
        self.c.execute(query, (experiment_id,))
        result = self.c.fetchone()
        return result[2] if result else {}

    def close_connection(self):
        """Close the database connection."""
        self.conn.close()
