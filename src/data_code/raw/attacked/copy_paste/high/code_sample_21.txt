        this.dest = dest;

    }

    public static void dijkstra(Graph graph, int source) {

        pq.add(new Edge(source, source, 0));
        distance[source] = 0; i < n; i++) {
            if (!visited[i]) {
                Node startNode = new Node(i);
                pq.offer(startNode);
                startNode.distance = 0;
            }
        }

        while (!pq.isEmpty()) {
            Node currNode = pq.poll();
            int currVertex = currNode.vertex;

            if (visited[currVertex])

        for (int i = 1; i < d.length; ++i)
