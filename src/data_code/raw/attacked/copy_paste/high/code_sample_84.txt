import java.util.concurrent.atomic.AtomicLong;

public class AtomicCounterBenchmark {
    private final int iterations;
    private final long startTimestamp;
    private final long endTimestamp;

    public AtomicCounterBenchmark(int iterations) {
        this.iterations = iterations;
        this.startTimestamp = System.currentTimeMillis();
    }

    public void run() throws InterruptedException {
        final AtomicLong atomicCounter = new AtomicLong(0);
        
        // Simulate concurrent updates
        for (int i = 0; i < iterations; i++) {
            atomicCounter.incrementAndGet();
        }
        
        // Record end timestamp
        endTimestamp = System.currentTimeMillis();
    }

    public double getAverageTimeTaken() {
        return (endTimestamp - startTimestamp) / ((double)iterations);
    }

    public static void main(String[] args) {
        try {
            AtomicCounterBenchmark benchmark = new AtomicCounterBenchmark(1000000);
            benchmark.run();

            System.out.println("Average time taken: " + benchmark.getAverageTimeTaken());
        } catch (InterruptedException e) {
            e.printStackTrace();
        }
    }
}
