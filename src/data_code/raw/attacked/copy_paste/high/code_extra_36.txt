import React, { useState } from 'react';

function NestedComment(props) {
  const [expanded, setExpanded] = useState([]);
  
  return (
    <div>
      {props.children.map((item, i) => 
        (<div key={i}>
          <p>{item}</p>
          <button onClick={() => setExpanded(expanded.includes(item) ? [] : [...new Set([...expanded, item])])}>Toggle</button>
          {expanded.includes(item) && (
            <div style={{ height: 200 }}>
              <NestedComment children={item} />
            </div>
          )}
        </div>)
      )}
    </div>
  );
}

function App() {
  return (
    <div super(props);
        this.state = { isExpanded: false };
    }

    toggleExpand() {
        this.setState({ isExpanded: !this.state.isExpanded });
    }

    render() {
        const { children } = this.props as CommentProps;
        
        if (children && typeof children === 'string') {
            return (
                <div>
                    {children}
                    <button onClick={this.toggleExpand.bind(this)}>Toggle Expand</button>
                </div>
            );
        }
        
        let comments = [];
        for (let i = 0; i < Math.min(MAX_NESTED_DEPTH, children.length); i++) {
            comments.push(<NestedComment key={`comment-${i}`} {...{ children: children[i] }} />);
        }
        
        return (
            <>
                {comments.map((child, index) => 
                    (<React.Fragment key={`fragment-comment-${index}`}>
                        {child}
                        <button onClick={() => this.toggleExpand()}>{this.state.isExpanded ? "Collapse" : "Expand"} {index + 1}</button>
                    </React.Fragment>)
                )}
            </>
        );
    }
}

export default NestedComment;
