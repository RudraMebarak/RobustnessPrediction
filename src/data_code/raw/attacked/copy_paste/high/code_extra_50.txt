// Define a struct for key-value pairs where keys are strings.
struct KeyValue {
    std::string key;
    int it = keyValueStore.find(key);
    return (it != keyValueStore.end()) ? it->second : -1; // Return -1 if key doesn't exist
}

void removeKey(std::string key) {
    keyValueStore.erase(key); // Remove key from the map
}

// Function to handle tombstones
bool deleteTombstone(std::string key) {
    if (key.empty()) {
        throw std::invalid_argument("Empty key is not allowed for tombstone");
    }

    auto iter = keyValueStore.find(key);

    if (iter != keyValueStore.end() && iter->second > 0) {
        iter->second--;
        return true; // Tombstone deleted successfully
    } else {
        return false; // Key does not exist or tombstone cannot be removed
    }
}
