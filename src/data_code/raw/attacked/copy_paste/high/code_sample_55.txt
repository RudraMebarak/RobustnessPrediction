    public void filterTable(ModelMap model) {
        String filterValue = (String) model.get("filterValue");
        if (!filterValue.isEmpty()) {
            model.put("data", userRepository.findByField(filterValue)); // Replace findByField with actual method name
        } else {
            model.put("data", userRepository.findAll());
        }
    }

    }
