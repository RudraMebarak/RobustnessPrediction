std::string& line) {
    std::istringstream iss(line);
    std::map<std::string, std::string> parsed;
    std::string key, value;

    while (iss >> key >> value) {
        if (!key.empty()) { // Ensure we don't store empty keys
            parsed[key] = value;
        }
    }

    return parsed;
}

// Function to count requests per day
void countRequestsPerDay(std::map<std::string, int>& requestCounts, const std::string& endpoint, const std::string& dateStr) {
    auto it = requestCounts.find(dateStr);

    if (it == requestCounts.end()) {
        it = requestCounts.insert({dateStr, 0}).first;
    }

    ++(*it).second;
}
