enum TokenType {
    TOKEN_PLUS,

    TOKEN_MULT,

    TOKEN_PAREN_OPEN,
    TOKEN_PAREN_CLOSE,
    TOKEN_EOF
};

struct Token {
    TokenType type;
    int value; // For numbers (integers)
    int lineNum; // Line number where this token occurs

    Token(TokenType t, int v) : type(t), value(v) {}
    Token(TokenType t, int v, int ln) : type(t), value(v), lineNum(ln) {}

    bool equals(Token other) const {
        return this->type == other.type && this->value == other.value;
    }
};
