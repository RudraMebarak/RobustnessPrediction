class TypeChecker {
    private Map<String, String> environment = new HashMap<>();

    public void checkType(String name) throws UnresolvedVariableException {
        if (environment.containsKey(name)) {
            System.out.println("Found in environment: " + environment.get(name));
        } else {
            throw new UnresolvedVariableException(name);
        }
    }

    class UnresolvedVariableException extends Exception {
        public UnresolvedVariableException(String name) {
            super("Unresolved variable: " + name);
        }
    }

    // Example usage
    public static void main(String[] args) {
        TypeChecker checker = new TypeChecker();
        
        try {
            checker.checkType("x");
        } catch (UnresolvedVariableException e) {
            System.out.println(e.getMessage());
        }

        // Add bindings to the environment
        checker.environment.put("x", "int");
        checker.environment.put("y", "boolean");

        try {
            checker.checkType("y"); // Should print "Found in environment: boolean"
            checker.checkType("z"); // Should throw an exception because 'z' is not defined
        } catch (UnresolvedVariableException e) {
            System.out.println(e.getMessage());
        }
    }

abstract class OperationExpression extends NumberExpression implements BinaryOperation {
    protected final Expression left;
    protected final Expression right;

    public OperationExpression(Expression left, Expression right) {
        this.left = left;
        this.right = right;
    }
}

abstract class BinaryOperation extends OperationExpression implements UnaryOperation {
    protected final Expression operand;

    public BinaryOperation(Expression left, Expression right, Expression operand) {
        super(left, right);
        this.operand = operand;
    }
}

abstract class UnaryOperation extends OperationExpression implements BinaryOperation {
    protected final Expression operand;

    public UnaryOperation(Expression operand) {
        this.operand = operand;
    }
}
