#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <pthread.h>

// Buffer size in bytes (1 byte per integer)
#define BUFFER_SIZE 1024

// Circular buffer structure
struct Buffer {
    int* buffer;
    int front;
    int rear;
    int capacity;
};

void initializeBuffer(struct Buffer* buffer) {
    buffer->buffer = (int*)malloc(BUFFER_SIZE * sizeof(int));
    if (buffer->buffer == NULL) {
        fprintf(stderr, "Memory allocation failed\n");
        exit(EXIT_FAILURE);
    }
    buffer->front = 0; // Position of the first available element
    buffer->rear = 0; // Position of the last used element before being placed back on stack
    buffer->capacity = BUFFER_SIZE; // Maximum number of elements in the buffer
}

void pushToBuffer(struct Buffer* buffer, int item) {
    pthread_mutex_lock(&mutex); // Acquire lock to ensure thread safety
    buffer->buffer[buffer->rear] = item; // Place new item at the current rear position
    buffer->rear = (buffer->rear + 1) % buffer->capacity; // Move rear pointer forward
    pthread_cond_signal(&notEmpty); // Notify waiting consumer threads
    pthread_mutex_unlock(&mutex); // Release lock after operation completed safely
}

#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>

#define BUFFER_SIZE 5

typedef struct {
    int data[BUFFER_SIZE];
    int head;
    int tail;
    pthread_mutex_t lock;
} Buffer;

Buffer buffer = {0};

void *producer(void *arg) {
    while (1) {
        pthread_mutex_lock(&buffer.lock);
        if (buffer.tail == BUFFER_SIZE - 1) {
            printf("Producer: Buffer full\n");
            pthread_cond_wait(&buffer.cond_full, &buffer.lock);
        }
        else {
            buffer.data[buffer.tail] = rand() % 100; // Simulate producing random data
            buffer.tail++;
            pthread_cond_signal(&buffer.cond_empty); // Signal consumer to start consuming
        }
        pthread_mutex_unlock(&buffer.lock);
        sleep(rand() % 3 + 1); // Simulate work time
    }
}

void *consumer(void *arg) {
    while (1) {
        pthread_mutex_lock(&buffer.lock);
        if (buffer.head == 0) {
            printf("Consumer: Buffer empty\n");
            pthread_cond_wait(&buffer.cond_empty, &buffer.lock);
        }
        else {
            int item = buffer.data[buffer.head];
            buffer.head--;
            pthread_cond_signal(&buffer.cond_full); // Signal producer to stop producing
            printf("Consumer: Consumed %d\n", item);
        }
        pthread_mutex_unlock(&buffer.lock);
        sleep(rand() % 2 + 1); // Simulate consumption time
    }
}

int main() {
    pthread_t t_producer, t_consumer;
    
    pthread_create(&t_producer, NULL, producer, NULL);
    pthread_create(&t_consumer, NULL, consumer, NULL);

    pthread_join(t_producer, NULL);
    pthread_join(t_consumer, NULL);

    return 0;
}
