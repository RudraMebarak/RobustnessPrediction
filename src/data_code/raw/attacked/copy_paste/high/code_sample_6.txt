def lsd_radix_sort(arr):
    max_val = max(arr)
    
    # Finding highest significant digit
    msd_max_value = (max_val - 1).bit_length()
    
    count = [0] * msd_max_value
    
    for num in arr:
        current_digit_index = (num // (1 << (msd_max_value - 1))) % 2 ** msd_max_value
        
        if count[current_digit_index]:
            return "Error: Not unique digits detected"
        
        count[current_digit_index] exp in range(1, 33):  # For 32-bit unsigned integer
        count = [0] * (max_val // exp + 1)
        
        for num in arr:
            index = (num // exp) % 10
            count[index] += 1
        
        for i in range(1, len(count)):
            count[i] += count[i - 1]
        
        output = [0] * len(arr)
        
        for num in reversed(arr):
            index = (num // exp) % 10
            output[count[index] - 1] = num
            count[index] -= 1
        
        arr.clear()
        arr.extend(output)
