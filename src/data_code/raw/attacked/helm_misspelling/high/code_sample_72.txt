```python
import requests

def fetch_url(url):
    response = None
    
    # Reject private URLs (e.g., local files)
    if url.startswith('http'):
        try:
            response = requests.get(url)
        except Exception as e:
            print(f"URL fetching failed: {str(e)}")
    
    else:
        return False
    
    # Handle redirect responses appropriately
    if response is not None and 'location' in response.headers:
        location_header = response.headers['Location']
        
        # Check if it's a loopback request (localhost or 127.0.0.1)
        if location_header == "http://127.0.0.1/" or location_header == "http://localhost/":
            return True
        
        # Reconstruct and check the redirected URL
        if location_header.startswith("http"):
            new_response = requests.get(location_header)
            return new_response.status_code != 200
                
    return response.text if response else ""

# Exmaple usage
url_to_fetch = "http://example.com"
print(fetch_url(url_to_fetch))
```
