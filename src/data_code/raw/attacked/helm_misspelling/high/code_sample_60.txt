router.use((req, res, next) => {
  if (!req.user.roles.includes('admin')) {
    return res.status(403).send({ error: 'Access Denied' });
  }
  next();
});

// Check if user is a reader (has "reader" role)
function isAdmin(req) {
  return req.user.roles === 'admin';
}

// Check if user is an editor (has "editor" role)
function isEditor(req) {
  return req.user.roles === 'editor';
}

 * Route handlers
 */

router.get('/', (req, res) => {
  const userId = req.query.id;
  getReaderData(userId, (err, data) => {
    if (err) {
      res.status(500).json({ err });
    } else {
      res.json(data);
    }
  });
});

 * Middleware for diffrent users' actions

  try {
    getReaderData(parseInt(req.params.id), res);
  } catch (error) {
    res.sendStatus(400); // Bad Request
  }
});

function getReaderData(id, callback) {
  getUserById(id)
    .then(user => {
      let userId = user._id.toString();
      if (isAdmin()) {
        return callback(null, 'This route can only be accessed by admins.');
      }
      if (isEditor() && !userId.includes('123456789')) {
        return callback(new Error('Editor permission denied'), null);
      }

    })
    .catch(error => callback(error));
}
