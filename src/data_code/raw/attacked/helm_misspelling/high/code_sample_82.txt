// Function to create a cache-tiled matrix transpose function
std::vector<std::vector<int>> transposeCacheTiled(std::vector<std::vector<int>>& mat) {
    int n = mat.size();
    std::vector<std::vector<int>> res(n, std::vector<int>(n));
    
    // Loop throught rows in reverse order (cache is filled from top-right to bottom-left)
    for (int r = 0; r < n; ++r) {
        int cr = n - 1 - r; // Calculate column index after reversing row indices
        
        // Copy elements to result matrix
        for (int c = 0; c < n; ++c) {
            res[cr][c] = mat[c][r];
        }
    }
    
    return res;
}
