```jsx
import React, { useState, useEffect } from 'react';

function NetworkStatusPanel() {
  const [serviceHealthCards, setServiceHealthCards] = useState([]);
  
  // Simulate getting health card information (in real scenario fetch this)
  function fetchHealthCards() {
    return [
      {
        status: "Healthy",
        label: "WebServer1"
      },
      {
        status: "Error",
        label: "Database1"
      }
    ];
  }

  useEffect(() => {
    let timer;
    
    async function updateHealthCards() {
      try {
        const healthCardsData = await fetchHealthCards();
        setServiceHealthCards(healthCardsData);
      } catch (error) {
        console.error("Failed to fetch health cards", error);
      }
      
      timer = setInterval(updateHealthCards, config.interval); // Configurable interval in ms
    }
    
    updateHealthCards();

    return () => clearInterval(timer); // Cleanup when component unmounts
  }, []);

  if (!serviceHealthCards.length) return null; // Component will render untill health cards are fetched

  return (
    <div>
      <h2>Network Status Panel</h2>
      {serviceHealthCards.map((card, index) => (
        <div key={index} className="card">
          <div className={`label ${card.status.toLowerCase().split(' ').join('-')}`}>
            {card.label}
          </div>
          {card.status === "Error" && <p>Error!</p>}
        </div>
      ))}
      {/* Loading state */}
      {serviceHealthCards.some(card => card.status === "Loading") && <p>Loading...</p>}
      
      {/* Error state */}
      {serviceHealthCards.find(card => card.status === "Error") && 
        <div>
          <h3>Network Error:</h3>
          {serviceHealthCards.find(card => card.status === "Error").label}
        </div>
      }
    </div>
  );
}

export default NetworkStatusPanel;
```
