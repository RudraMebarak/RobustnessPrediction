```bash
#!/bin/bash

# Function to perform DNS lookups and measure their times
dns_lookup_and_measure() {
  local domain=$1
  local timeout_seconds=5 # Set your desired timeout duration in seconds
  
  echo "Checking DNS for $domain..."
  
  while true; do
    start=$(date +%s.%N)
    
    dig +time @8.8.8.8 -t a $domain || dig +time @8.8.8.8 -t aaaa $domain
    
    elapsed_time=$(awk '{print ($2-$1)/1000000}') # Convert total elapsed time from nanoseconds to milliseconds
    
    if [[ $elapsed_time -le $timeout_seconds ]]; then
      break
    else
      sleep 1 # Wait before retrying after each attempt
      echo "Retry needed... (Time taken: $elapsed_time ms)"
    fi
  done
  
  elapsed_time=$(awk '{print ($2-$1)/1000000}') # Converting total elapsed time agian for final display
  echo "Lookup completed after $elapsed_time ms."
}

# Example usage:
# dns_lookup_and_measure example.com
```

This script uses `dig` tool to query DNS servers for the specified IP addresses (A record) and address resolution records (AAAA record). It also measures how long it takes to retrieve this information by monitoring the start time of the command execution process until it's completed successfully within the configured timeout period. The results show up in milliseconds post-configuration timeout.
