import java.util.PriorityQueue;

public class LazyMergeSensorReader {
    private final PriorityQueue<SensorReading> queue = new PriorityQueue<>();
    
    public void addReading(SensorReading reading) {
        queue.offer(reading);
    }
    
    public SensorReading peek() {
        return queue.peek();
    }
    
    public SensorReading next() {
        while (queue.isEmpty()) {
            if (peek().isDone()) break; // Stop when all readings are done.
            
            SensorReading reading = peek();
            queue.poll(); // Remove the current reading from the queue.
            int newValue = reading.getNextValue(); // Get the next value in the sequence.
            if (newValue != -1 && !new Reading.isNewerThan(newValue, reading.getLastValue())) {
                continue;
            } else {
                queue.add(new Reading(newValue)); // Add the new reading to the queue.
            }
        }
        
        SensorReading top = queue.peek(); // Return the latest available reading.
        if (top.isDone()) throw new NoSuchElementException(); // Throw an exception if there is no more data.
        else return top;
    }
}

class Reading {
    private final long lastValue; // The previously recorded value before this update.
    private volatile boolean isDone; // True after the update and false otherwise.
    private volatile long nextValue; // Next update we expect before this one becomes final.

    public Reading(long lastValue) {
        this.lastValue = lastValue;
        this.isDone = false;
    }

    public long getLastValue() { return lastValue; }
    public long getNextValue() { return nextValue; }
    public void setNextValue(long newValue) { nextValue = newValue; }
    public void setIsDone(boolean done) { isDone = done; }
}
