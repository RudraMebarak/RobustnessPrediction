  # mean imputation
  data$col1_imputed <- ifelse(is.na(data$col1), mean(data$col1), data$col1)

  # interpolation (assuming linear relation between non-missing points)

#### step 4: outlier detection
use statistical methods like z-scores or iqrs:

```r
z_scores <- apply(data[,c('col1','col2')], 1, zscore)
outliers_z <- which(abs(z_scores[,1]) > 3 | abs(z_scores[,2]) > 3)
data$outliers <- ifelse(outliers_z==1|outliers_z==2,'true','false')
```

```python
z_scores = ((data.iloc[:,0] - data.iloc[:,0].mean()) / data.iloc[:,0].std())
outliers_z = [x for x in range(len(data)) if abs(z_scores.loc[x][1])>3]
data['outliers'] = 'true' * len(outliers_z) + 'false'
```
